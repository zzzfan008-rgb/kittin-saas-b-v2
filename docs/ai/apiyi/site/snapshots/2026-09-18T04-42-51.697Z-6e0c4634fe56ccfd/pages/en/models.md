export const ModelSearch = () => {
  const [q, setQ] = useState('');
  const query = q.trim().toLowerCase();
  const all = query ? MODELS.filter(m => m.n.toLowerCase().includes(query)) : [];
  const hits = all.slice(0, 30);
  return <div className="mp-search">
      <div className="mp-search-bar">
        <p className="mp-search-label">Search Now</p>
        <input value={q} onChange={e => setQ(e.target.value)} placeholder="Search model name, e.g. gpt-5.2 / claude / nano-banana" className="mp-search-input" />
      </div>
      {query.length > 0 && all.length === 0 && <p className="mp-search-empty">No matching model found</p>}
      {query.length > 0 && all.length > 0 && <div>
          <p className="mp-search-meta">{all.length > 30 ? '' + all.length + ' matches' + ' — showing first 30, keep typing to narrow down' : '' + all.length + ' matches'}</p>
          <table>
            <thead>
              <tr>
                <th>Model</th>
                <th>Billing</th>
                <th>Input</th>
                <th>Output</th>
                <th>Cache read</th>
                <th>Vendor</th>
              </tr>
            </thead>
            <tbody>
              {hits.map(m => <tr key={m.n}>
                  <td className="mp-mono">{m.href ? <a href={m.href}>{m.n}</a> : m.n}{m.t ? ' †' : ''}</td>
                  <td>{m.b}</td>
                  <td>{m.i}</td>
                  <td>{m.o}</td>
                  <td>{m.c}</td>
                  <td><a href={'#' + m.a}>{m.v}</a></td>
                </tr>)}
            </tbody>
          </table>
        </div>}
    </div>;
};
